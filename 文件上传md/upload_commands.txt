# Git Upload Commands - Run these one by one in PowerShell
# Replace YOUR_GITHUB_REPO_URL with your actual repository URL

# 1. Configure Git user (replace with your info)
git config --global user.name "YourGitHubUsername"
git config --global user.email "your.email@example.com"

# 2. Add remote repository (replace with your repo URL)
git remote add origin https://github.com/username/repo.git

# 3. Verify remote
git remote -v

# 4. Add all files (respecting .gitignore)
git add .

# 5. Check what will be committed
git status

# 6. Commit changes
git commit -m "Update: First commit after migrating to new computer - Add latest analysis results and documentation - Exclude large files via .gitignore"

# 7. Create and push to update-1119 branch
git checkout -b update-1119
git push -u origin update-1119

# If push fails due to authentication, you'll need to:
# - Use Personal Access Token as password (not your GitHub password)
# - Or configure SSH keys

# After successful push:
# - Visit your GitHub repository
# - Create a Pull Request from update-1119 to main
# - Merge the PR
